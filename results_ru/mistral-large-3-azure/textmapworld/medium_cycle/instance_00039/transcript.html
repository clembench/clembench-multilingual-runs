from collections import deque

# Read input
input_lines = [line.strip() for line in sys.stdin if line.strip()]
start_room = input_lines[0].split(": ")[1]
directions = input_lines[1].split(": ")[1].split(", ")
directions = [d.strip() for d in directions]

# Build adjacency list
adj = {}
visited = set()
queue = deque()

# Start BFS
queue.append((start_room, []))
visited.add(start_room)

while queue:
current_room, path = queue.popleft()
print(f"ИДТИ: {path[0]}" if path else "ГОТОВО", flush=True)

# Read next input
response = input().strip()
if response == "ГОТОВО":
break

next_room = response.split(": ")[1]
next_directions = input().split(": ")[1].split(", ")
next_directions = [d.strip() for d in next_directions]

# Add neighbors to the queue
for direction in next_directions:
if direction not in adj.get(next_room, []):
adj.setdefault(next_room, []).append(direction)
opposite = {"север": "юг", "юг": "север", "восток": "запад", "запад": "восток"}[direction]
neighbor_room = f"Room_{len(visited)}" # Placeholder, actual room name comes from input
adj.setdefault(neighbor_room, []).append(opposite)
if neighbor_room not in visited:
visited.add(neighbor_room)
queue.append((neighbor_room, [direction]))
