from collections import deque

def solve_maze(start_room, get_neighbors_func):
# Хранит посещенные комнаты и путь к ним
visited = {}
queue = deque()

# Начинаем с начальной комнаты
queue.append((start_room, []))
visited[start_room] = True

while queue:
current_room, path = queue.popleft()

# Получаем соседей текущей комнаты
neighbors = get_neighbors_func(current_room)

for direction, neighbor in neighbors.items():
if neighbor not in visited:
visited[neighbor] = True
new_path = path + [direction]
queue.append((neighbor, new_path))

# Если мы нашли все комнаты, возвращаем путь
if len(visited) == len(get_all_rooms(get_neighbors_func)):
return new_path

return [] # Если лабиринт не полностью исследован

# Простая функция для сбора всех комнат (неэффективна для больших лабиринтов)

while queue:
room = queue.popleft()
if room in visited:
continue
visited.add(room)
rooms.add(room)
neighbors = get_neighbors_func(room)
for neighbor in neighbors.values():
if neighbor not in visited:
queue.append(neighbor)

return rooms
