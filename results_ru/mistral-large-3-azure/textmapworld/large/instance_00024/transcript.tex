\texttt{from collections import deque} \\
\\ 
\texttt{def solve\_maze(start\_room, get\_neighbors\_func):} \\
\texttt{    \# Хранит посещенные комнаты и путь к ним} \\
\texttt{    visited = \{\}} \\
\texttt{    queue = deque()} \\
\\ 
\texttt{    \# Начинаем с начальной комнаты} \\
\texttt{    queue.append((start\_room, {[}{]}))} \\
\texttt{    visited{[}start\_room{]} = True} \\
\\ 
\texttt{    while queue:} \\
\texttt{        current\_room, path = queue.popleft()} \\
\\ 
\texttt{        \# Получаем соседей текущей комнаты} \\
\texttt{        neighbors = get\_neighbors\_func(current\_room)} \\
\\ 
\texttt{        for direction, neighbor in neighbors.items():} \\
\texttt{            if neighbor not in visited:} \\
\texttt{                visited{[}neighbor{]} = True} \\
\texttt{                new\_path = path + {[}direction{]}} \\
\texttt{                queue.append((neighbor, new\_path))} \\
\\ 
\texttt{                \# Если мы нашли все комнаты, возвращаем путь} \\
\texttt{                if len(visited) == len(get\_all\_rooms(get\_neighbors\_func)):} \\
\texttt{                    return new\_path} \\
\\ 
\texttt{    return {[}{]}  \# Если лабиринт не полностью исследован} \\
\\ 
\texttt{def get\_all\_rooms(get\_neighbors\_func):} \\
\texttt{    \# Простая функция для сбора всех комнат (неэффективна для больших лабиринтов)} \\
\texttt{    rooms = set()} \\
\texttt{    queue = deque({[}"Спальня"{]})} \\
\texttt{    visited = set()} \\
\\ 
\texttt{    while queue:} \\
\texttt{        room = queue.popleft()} \\
\texttt{        if room in visited:} \\
\texttt{            continue} \\
\texttt{        visited.add(room)} \\
\texttt{        rooms.add(room)} \\
\texttt{        neighbors = get\_neighbors\_func(room)} \\
\texttt{        for neighbor in neighbors.values():} \\
\texttt{            if neighbor not in visited:} \\
\texttt{                queue.append(neighbor)} \\
\\ 
\texttt{    return rooms} \\
