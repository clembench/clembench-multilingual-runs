from collections import deque


class MazeSolver:
def __init__(self, maze):
self.maze = maze
self.visited = set()
self.queue = deque()
self.parent = {}
self.start = None
self.end = None

self._find_start_and_end()

def _find_start_and_end(self):
for i in range(len(self.maze)):
for j in range(len(self.maze[i])):
if self.maze[i][j] == 'S':
self.start = (i, j)
elif self.maze[i][j] == 'E':
self.end = (i, j)

def _is_valid_move(self, x, y):
return 0 <= x < len(self.maze) and 0 <= y < len(self.maze[0]) and self.maze[x][y] != '#' and (x, y) not in self.visited

def solve(self):
if not self.start or not self.end:
return None

self.queue.append(self.start)
self.visited.add(self.start)

directions = [(-1, 0), (1, 0), (0, -1), (0, 1)]

while self.queue:
current = self.queue.popleft()

if current == self.end:
return self._reconstruct_path(current)

for dx, dy in directions:
next_x, next_y = current[0] + dx, current[1] + dy
if self._is_valid_move(next_x, next_y):
self.visited.add((next_x, next_y))
self.parent[(next_x, next_y)] = current
self.queue.append((next_x, next_y))

return None

def _reconstruct_path(self, current):
path = []
while current != self.start:
path.append(current)
current = self.parent[current]
path.append(self.start)
path.reverse()
return path
