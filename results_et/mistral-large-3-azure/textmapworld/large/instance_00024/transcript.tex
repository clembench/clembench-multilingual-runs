\texttt{def bfs(graph, start, goal):} \\
\texttt{    queue = deque({[}(start, {[}start{]}){]})} \\
\texttt{    visited = set({[}start{]})} \\
\\ 
\texttt{    while queue:} \\
\texttt{        current, path = queue.popleft()} \\
\\ 
\texttt{        if current == goal:} \\
\texttt{            return path} \\
\\ 
\texttt{        for neighbor in graph{[}current{]}:} \\
\texttt{            if neighbor not in visited:} \\
\texttt{                visited.add(neighbor)} \\
\texttt{                queue.append((neighbor, path + {[}neighbor{]}))} \\
\\ 
\texttt{    return None} \\
