class Room:
def __init__(self, name):
self.name = name
self.neighbors = {} # Direction: Room
self.visited = False

def add_neighbor(self, direction, room):
self.neighbors[direction] = room

def get_unvisited_neighbors(self):
return [dir for dir, room in self.neighbors.items() if not room.visited]

def __str__(self):
return self.name

def build_house():
# Create rooms
kasvuhoone = Room("Kasvuhoone")
tookoda = Room("Töökoda")
kodujousaal = Room("Kodujõusaal")
kodukontor = Room("Kodukontor")

# Connect rooms
kasvuhoone.add_neighbor("lõuna", tookoda)
kasvuhoone.add_neighbor("põhja", kodujousaal)

tookoda.add_neighbor("põhja", kasvuhoone)

kodujousaal.add_neighbor("lõuna", kasvuhoone)
kodujousaal.add_neighbor("põhja", kodukontor)
kodujousaal.add_neighbor("lääne", kodukontor)

kodukontor.add_neighbor("lõuna", kodujousaal)
kodukontor.add_neighbor("ida", kodujousaal)

return kasvuhoone

def navigate_house(start_room):

path = []
