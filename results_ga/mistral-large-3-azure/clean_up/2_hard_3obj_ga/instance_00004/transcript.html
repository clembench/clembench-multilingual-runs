Define the function and its derivative

def f(x): return x**2

def df(x): return 2*x

Gradient Descent Algorithm

def gradient_descent(starting_point, learning_rate, num_iterations): x = starting_point x_history = [x] f_history = [f(x)]

for _ in range(num_iterations):
    gradient = df(x)
    x = x - learning_rate * gradient
    x_history.append(x)
    f_history.append(f(x))

return x_history, f_history
